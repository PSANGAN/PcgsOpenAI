
// Manual Schema Initialization Bean
@Configuration
public class DatabaseConfig {

    @Bean
    @ConditionalOnProperty(
        prefix = "spring.ai.chat.memory.jdbc",
        name = "initialize-schema",
        havingValue = "true",
        matchIfMissing = true
    )
    public CommandLineRunner initSchema(DataSource dataSource) {
        return args -> {
            try (Connection conn = dataSource.getConnection();
                 Statement stmt = conn.createStatement()) {

                // Check if table exists
                try {
                    stmt.execute("SELECT 1 FROM SPRING_AI_CHAT_MEMORY LIMIT 1");
                    System.out.println("✓ SPRING_AI_CHAT_MEMORY table already exists");
                } catch (SQLException e) {
                    // Table doesn't exist, create it
                    System.out.println("Creating SPRING_AI_CHAT_MEMORY table...");

                    String sql = """
                        CREATE TABLE SPRING_AI_CHAT_MEMORY (
                            conversation_id VARCHAR(36) NOT NULL,
                            content LONGVARCHAR NOT NULL,
                            type VARCHAR(10) NOT NULL,
                            "timestamp" TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL
                        );

                        CREATE INDEX SPRING_AI_CHAT_MEMORY_CONVERSATION_ID_TIMESTAMP_IDX
                        ON SPRING_AI_CHAT_MEMORY(conversation_id, "timestamp" DESC);

                        ALTER TABLE SPRING_AI_CHAT_MEMORY
                        ADD CONSTRAINT TYPE_CHECK
                        CHECK (type IN ('USER', 'ASSISTANT', 'SYSTEM', 'TOOL'));
                        """;

                    stmt.execute(sql);
                    System.out.println("✓ SPRING_AI_CHAT_MEMORY table created successfully");
                }
            }
        };
    }
}
